# EMOM — piano di implementazione

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** l'EMOM nell'app — riquadro nell'editor, orologio che corre in allenamento, import, recap/storico ed esportazione.

**Architecture:** un EMOM è una SUPERSERIE (lib/superserie: flag `insiemeAlPrecedente` sul successivo) con `emom = { ogniSec, minuti }` sul primo esercizio. Le serie di ogni esercizio si calcolano (`allineaEmom`, lib/emom) e si scrivono nello schema, così recap, volume, storico ed export le leggono come serie normali. In allenamento il blocco si mostra con `CardEmom`, un orologio calcolato dall'istante di partenza salvato nella sessione.

**Tech Stack:** React 19 + Vite, JS senza TypeScript, prove con `node --test` (`npm run test`), lint con `npm run lint`.

**Spec:** `docs/superpowers/specs/2026-11-03-emom-design.md`

```js
// src/lib/emom.js
// ---------------------------------------------------------------------------
// L'EMOM ("every minute on the minute"): a ogni intervallo si parte con una
// serie, e quello che resta dell'intervallo è il recupero. È una SUPERSERIE
// (lib/superserie) con le impostazioni sul primo esercizio:
//
//   Esercizio.emom = { ogniSec: 60, minuti: 12 }   // minuti null = a oltranza
//
// Più esercizi si alternano un intervallo per uno (A, B, A, B…): è l'ordine
// del giro. Le serie di ognuno si CALCOLANO (allineaEmom) ma si scrivono nello
// schema: recap, volume, storico ed esportazione le leggono come serie normali.
// Spec: docs/superpowers/specs/2026-11-03-emom-design.md
// ---------------------------------------------------------------------------
import { faseVuota, formattaSecondi, normalizzaSchema } from './schema.js'
import { blocchi } from './superserie.js'

export const OGNI_PROPOSTI = [30, 45, 60, 90, 120, 180]
export const MINUTI_MAX = 60
// I secondi di "Pronti… 3, 2, 1" fra "Avvia EMOM" e il primo intervallo.
export const PRONTI_SEC = 10

/** Quanti intervalli: il pezzo di minuto che avanza non si fa. null = a oltranza. */
export function intervalli(emom) {
  if (!emom || emom.minuti == null) return null
  return Math.max(1, Math.floor((emom.minuti * 60) / emom.ogniSec))
}

/** Le serie dell'esercizio k (da 0) su n: i primi prendono l'intervallo in più. */
export function serieEmom(emom, n, k) {
  const tot = intervalli(emom)
  return tot == null ? null : Math.max(0, Math.ceil((tot - k) / n))
}

/** Gli intervalli (da 1) dell'esercizio k su n: 1·3·5… null = a oltranza. */
export function minutiDi(emom, n, k) {
  const tot = intervalli(emom)
  if (tot == null) return null
  const out = []
  for (let m = k; m < tot; m += n) out.push(m + 1)
  return out
}

/** All'intervallo m (da 0) tocca l'esercizio k del blocco, alla sua serie j. */
export function esercizioDelMinuto(n, m) {
  return { k: m % n, j: Math.floor(m / n) }
}

/**
 * Dov'è l'orologio, calcolato dall'istante di partenza (`inizio`, ms): regge
 * il telefono in tasca e "Esci" → "Riprendi". `pronti` = secondi del conto
 * alla rovescia iniziale (0 quando si è partiti), `minuto` = l'intervallo in
 * corso (da 0), `restanteSec` = quanto gli resta.
 */
export function statoOrologio(inizio, emom, adesso) {
  const t = (adesso - inizio) / 1000 - PRONTI_SEC
  if (t < 0) return { pronti: Math.ceil(-t), minuto: 0, restanteSec: emom.ogniSec, finito: false }
  const minuto = Math.floor(t / emom.ogniSec)
  const tot = intervalli(emom)
  if (tot != null && minuto >= tot) return { pronti: 0, minuto: tot, restanteSec: 0, finito: true }
  return { pronti: 0, minuto, restanteSec: Math.ceil(emom.ogniSec * (minuto + 1) - t), finito: false }
}

/** "EMOM 12' · ogni 1'"; a oltranza, o fermato prima, fin dove si è arrivati. */
export function testoEmom(emom) {
  const ogni = `ogni ${formattaSecondi(emom.ogniSec)}`
  const tot = intervalli(emom)
  const fatti = emom.minutiFatti
  if (fatti != null && (tot == null || fatti < tot)) {
    return `EMOM · ${ogni} · ${emom.ogniSec === 60 ? `fino al ${fatti}° minuto` : `${fatti} intervalli`}`
  }
  return tot == null ? `EMOM · ${ogni} · finché ce la faccio` : `EMOM ${emom.minuti}' · ${ogni}`
}

/**
 * Gli esercizi di una SCHEDA con ogni blocco EMOM in ordine: `emom` solo sul
 * primo (preso da chiunque del blocco ce l'abbia), le serie calcolate, una
 * fase sola, niente recupero né settimane diverse. Chi non è in un EMOM
 * resta lo stesso oggetto. Si lancia dopo ogni modifica della lista e in
 * normalizzaScheda: una serie fuori conto non resta mai.
 */
export function allineaEmom(esercizi) {
  const out = [...(esercizi || [])]
  for (const b of blocchi(out)) {
    const emom = b.indici.map((i) => out[i].emom).find(Boolean) || null
    b.indici.forEach((i, k) => {
      let e = out[i]
      if (k > 0 && e.emom) {
        const { emom: _, ...resto } = e
        e = resto
      }
      if (emom) {
        if (k === 0 && e.emom !== emom) e = { ...e, emom }
        const serie = serieEmom(emom, b.indici.length, k)
        const s = normalizzaSchema(e.schemaBase)
        const f = s.fasi[0] || faseVuota()
        if (s.fasi.length !== 1 || f.serie !== serie || s.recuperoSec != null || e.variaPerSettimana) {
          e = { ...e, variaPerSettimana: false, schemaBase: { ...s, fasi: [{ ...f, serie }], recuperoSec: null } }
        }
      }
      out[i] = e
    })
  }
  return out
}

/** L'EMOM dell'esercizio i: le impostazioni, quanti sono (n) e lui chi è (k). */
export function posizioneEmom(esercizi, i) {
  const blocco = blocchi(esercizi).find((b) => b.indici.includes(i))
  const emom = blocco && esercizi[blocco.inizio]?.emom
  return emom ? { emom, n: blocco.indici.length, k: blocco.indici.indexOf(i), blocco } : null
}

/** "Sciogli": il blocco dell'esercizio `id` torna esercizi normali, slegati. */
export function sciogliEmom(esercizi, id) {
  const i = esercizi.findIndex((e) => e.id === id)
  const blocco = blocchi(esercizi).find((b) => b.indici.includes(i))
  if (!blocco) return esercizi
  return esercizi.map((e, k) => {
    if (!blocco.indici.includes(k)) return e
    const { emom: _, ...resto } = e
    return { ...resto, insiemeAlPrecedente: false }
  })
}
```

```js
/**
 * L'EMOM a oltranza: a ogni intervallo nuovo l'esercizio di turno ha la sua
 * serie in più (da dare). Fino all'intervallo `minuto` (da 0) compreso.
 * Niente da aggiungere: la stessa lista.
 */
export function serieEmomFinoA(esercizi, blocco, minuto) {
  const n = blocco.indici.length
  let cambiato = false
  const out = esercizi.map((e, i) => {
    const k = blocco.indici.indexOf(i)
    if (k === -1) return e
    const servono = minuto >= k ? Math.floor((minuto - k) / n) + 1 : 0
    if (e.sets.length >= servono) return e
    cambiato = true
    return { ...e, sets: [...e.sets, ...Array.from({ length: servono - e.sets.length }, () => ({ colore: null }))] }
  })
  return cambiato ? out : esercizi
}

/**
 * Fine dell'EMOM (finito, "Ferma" o "Non ce la faccio più"): valgono i primi
 * `fatti` intervalli. Le serie oltre si tolgono; quelle non date valgono fatte
 * come da piano, senza sforzo (`'fatto'`, come il cardio).
 */
export function chiudiEmom(esercizi, blocco, fatti) {
  const n = blocco.indici.length
  return esercizi.map((e, i) => {
    const k = blocco.indici.indexOf(i)
    if (k === -1) return e
    const quante = fatti > k ? Math.ceil((fatti - k) / n) : 0
    return {
      ...e,
      ...(k === 0 ? { emom: { ...e.emom, minutiFatti: fatti } } : {}),
      sets: e.sets.slice(0, quante).map((s, j) => (s.colore ? s : serieChiusa(e.schema, j, 'fatto'))),
    }
  })
}
```

```jsx
// src/components/CardEmom.jsx
import { useEffect, useRef, useState } from 'react'
import { esercizioDelMinuto, intervalli, statoOrologio, testoEmom } from '../lib/emom'
import { fasiDi, formattaCarico, formattaRip, formattaSecondi } from '../lib/schema'
import { COLORI } from '../lib/session'
import { beep, sbloccaAudio } from '../hooks/useRestTimer'

const SFORZI = ['verde', 'giallo', 'rosso']

// Il blocco EMOM in allenamento (lib/emom): l'orologio va da solo, calcolato
// dall'istante di partenza salvato nella sessione (`emom.inizio`), così regge
// lo schermo spento e "Esci" → "Riprendi". Lo sforzo si dà toccando il
// pallino, quando si vuole: l'orologio non si ferma mai. Un intervallo non
// dato resta in sospeso; a fine EMOM vale fatto come da piano (chiudiEmom).
export default function CardEmom({ esercizi, blocco, attiva, onAvvia, onColore, onCresci, onChiudi }) {
  const emom = esercizi[blocco.inizio].emom
  const n = blocco.indici.length
  const tot = intervalli(emom)
  const finito = emom.minutiFatti != null
  const corre = emom.inizio != null && !finito
  const [adesso, setAdesso] = useState(() => Date.now())
  useEffect(() => {
    if (!corre) return
    const t = setInterval(() => setAdesso(Date.now()), 250)
    return () => clearInterval(t)
  }, [corre])
  const stato = corre ? statoOrologio(emom.inizio, emom, adesso) : null

  // A ogni intervallo nuovo: bip (se acceso), vibrazione e, a oltranza, la
  // serie in più. Riprendendo a metà non suona: l'intervallo era già iniziato.
  const ultimo = useRef(null)
  const minuto = stato && !stato.pronti ? stato.minuto : null
  useEffect(() => {
    if (minuto == null) return
    if (stato.finito) {
      onChiudi(tot)
      return
    }
    if (ultimo.current === minuto) return
    if (ultimo.current !== null || minuto === 0) {
      beep()
      navigator.vibrate?.(200)
    }
    ultimo.current = minuto
    if (tot == null) onCresci(minuto)
  }, [minuto, stato?.finito]) // eslint-disable-line react-hooks/exhaustive-deps

  // Il pallino scelto per dare lo sforzo; se no quello dell'intervallo in corso.
  const [scelto, setScelto] = useState(null)
  const visti = finito ? emom.minutiFatti : tot ?? (minuto ?? 0) + 1
  const diMinuto = (m) => {
    const { k, j } = esercizioDelMinuto(n, m)
    return { i: blocco.indici[k], j, m }
  }
  const quiOra = minuto != null ? diMinuto(minuto) : null
  const bersaglio = scelto || quiOra
  const descrivi = (i) => {
    const f = fasiDi(esercizi[i].schema)[0] || {}
    return [esercizi[i].nome, formattaRip(f.rip) && `×${formattaRip(f.rip)}`, formattaCarico(f.carico)].filter(Boolean).join(' · ')
  }
  const dopo = minuto != null && (tot == null || minuto + 1 < tot) ? diMinuto(minuto + 1) : null

  return (
    <div className={'card emom-card' + (attiva ? '' : ' non-attiva')} inert={!attiva}>
      <div className="emom-titolo">{testoEmom(emom)}</div>

      {!corre && !finito && (
        <>
          <ul className="emom-piano">
            {blocco.indici.map((i) => (
              <li key={i}>{descrivi(i)}</li>
            ))}
          </ul>
          <button
            className="btn btn-accent btn-block"
            onClick={() => {
              sbloccaAudio()
              onAvvia()
            }}
          >
            Avvia EMOM
          </button>
        </>
      )}

      {stato && stato.pronti > 0 && (
        <div className="emom-orologio">
          <span className="emom-minuto">Pronti…</span>
          <span className="emom-conto">{stato.pronti}</span>
        </div>
      )}

      {stato && !stato.pronti && quiOra && (
        <>
          <div className={'emom-orologio' + (stato.restanteSec <= 3 ? ' ultimi' : '')}>
            <span className="emom-minuto">
              {emom.ogniSec === 60 ? 'Minuto' : 'Intervallo'} {minuto + 1}
              {tot != null && ` di ${tot}`}
            </span>
            <span className="emom-conto">{formattaSecondi(stato.restanteSec).replace(/^(\d+)"$/, '0:$1')}</span>
          </div>
          <div className="emom-ora">{descrivi(quiOra.i)}</div>
          {dopo && <div className="emom-dopo">Poi: {descrivi(dopo.i)}</div>}
        </>
      )}

      {finito && <div className="emom-finito">EMOM finito ✓</div>}

      {(corre || finito) && (
        <>
          <div className="emom-pallini" role="group" aria-label="Intervalli">
            {Array.from({ length: visti }, (_, m) => {
              const x = diMinuto(m)
              const colore = esercizi[x.i].sets[x.j]?.colore || ''
              const acceso = bersaglio && bersaglio.m === m
              return (
                <button
                  key={m}
                  type="button"
                  className={'set-dot ' + colore + (acceso ? ' acceso' : '')}
                  aria-label={`Intervallo ${m + 1}: ${esercizi[x.i].nome}`}
                  onClick={() => setScelto(x)}
                >
                  {m + 1}
                </button>
              )
            })}
          </div>
          {bersaglio && esercizi[bersaglio.i].sets[bersaglio.j] && (
            <div className="emom-sforzo">
              <span className="muted">
                Com'è andata? {esercizi[bersaglio.i].nome} · {bersaglio.m + 1}°
              </span>
              <div className="effort-buttons">
                {SFORZI.map((c) => (
                  <button
                    key={c}
                    className={'effort ' + c}
                    onClick={() => {
                      onColore(bersaglio.i, bersaglio.j, c)
                      setScelto(null)
                    }}
                  >
                    {COLORI[c].label}
                  </button>
                ))}
              </div>
            </div>
          )}
        </>
      )}

      {corre && (
        <button
          className="btn btn-block"
          style={{ marginTop: 12 }}
          onClick={() => {
            if (tot != null && !confirm('Fermare l’EMOM? Valgono gli intervalli fatti.')) return
            onChiudi(stato?.pronti ? 0 : minuto)
          }}
        >
          {tot == null ? 'Non ce la faccio più' : 'Ferma'}
        </button>
      )}
    </div>
  )
}
```

```js
function emomDa(t) {

  let ogniSec = (m[1] ? parseInt(m[1], 10) : 1) * 60
```
